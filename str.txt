#!/usr/bin/env python3

import sys
from unicorn import *
from unicorn.arm_const import *
from udbserver import udbserver

class ARMEmulator:
    def __init__(self, text_path, data_path, firmware_path):
        self.text_binary_path = text_path
        self.data_binary_path = data_path
        self.firmware_binary_path = firmware_path
        self.uc = Uc(UC_ARCH_ARM, UC_MODE_ARM)

        # Memory layout - match linker script addresses
        self.CODE_BASE = 0x10000   # .text at 0x10000
        self.DATA_BASE = 0x20000   # .data at 0x20000
        self.UART_BASE = 0x80000
        self.UART_SIZE = 0x1000
        self.FW_BASE = 0xC0000
        self.FW_SIZE = 0x10000
        self.STACK_BASE = 0x40000  # Stack
        self.STACK_SIZE = 0x10000

        self.UART_PRINT_ADDR = 0x00010070
        self.START_ADDR = 0x00010d38

        self.uart_calls = []

    def setup_memory(self):
        """Map memory regions for code, data, and stack"""
        # Map sections according to linker script
        self.uc.mem_map(self.CODE_BASE, 0x10000, UC_PROT_READ | UC_PROT_EXEC)

        # Map the rest here!
        self.uc.mem_map(self.DATA_BASE, 0x10000, UC_PROT_READ | UC_PROT_EXEC)
        self.uc.mem_map(self.UART_BASE, self.UART_SIZE, UC_PROT_READ | UC_PROT_WRITE)
        self.uc.mem_map(self.FW_BASE, self.FW_SIZE, UC_PROT_READ | UC_PROT_EXEC)
        self.uc.mem_map(self.STACK_BASE, self.STACK_SIZE, UC_PROT_READ | UC_PROT_WRITE)

        print(f"[+] Memory mapped:")
        print(f"    Code:     0x{self.CODE_BASE:08x} - 0x{self.CODE_BASE + 0x10000:08x}")

    def load_binary(self):
        """Load the ARM binary sections into emulated memory"""
        try:
            # Load .text section (linked with resolved relocations)
            with open(self.text_binary_path, 'rb') as f:
                text_data = f.read()
            self.uc.mem_write(self.CODE_BASE, text_data)
            print(f"[+] .text: {len(text_data)} bytes at 0x{self.CODE_BASE:08x}")

            # Load .data section
            with open(self.data_binary_path, 'rb') as f:
                text_data = f.read()
            self.uc.mem_write(self.DATA_BASE, text_data)
            print(f"[+] .data: {len(text_data)} bytes at 0x{self.DATA_BASE:08x}")
            
            # Load .data section
            with open(self.firmware_binary_path, 'rb') as f:
                text_data = f.read()
            self.uc.mem_write(self.FW_BASE, text_data)
            print(f"[+] .firmware: {len(text_data)} bytes at 0x{self.FW_BASE:08x}")

        except Exception as e:
            print(f"[-] Error loading binary sections: {e}")
            sys.exit(1)

    def setup_registers(self):
        """Initialize ARM registers"""
        # Set stack pointer here!
        self.uc.reg_write(UC_ARM_REG_SP, self.STACK_BASE + self.STACK_SIZE)

    def hook_code(self, uc, address, size, user_data):
        # print(f"[EXEC] 0x{address:08x}")
        pass

    def hook_fix_uart(self, uc, address, size, user_data):
        # print("Fixing uart_print")
        self.uc.reg_write(UC_ARM_REG_R0, 0x0)
    
    def hook_uart_print(self, uc, address, size, user_data):
        ptr = self.uc.reg_read(UC_ARM_REG_R1)
        raw = bytes(self.uc.mem_read(ptr, 256))
        val = raw.split(b"\x00")[0].decode("utf-8")        
        print(f"[UART]> {val}")

    

    def emulate(self):
        """Start ARM emulation"""
        print(f"[+] Starting emulation from 0x{self.START_ADDR:08x}")

        # Add hooks here!
        self.uc.hook_add(UC_HOOK_CODE, self.hook_code)  # Enable for debugging
        
        self.uc.hook_add(UC_HOOK_CODE, self.hook_fix_uart, begin=0x10090, end=0x10090)
        self.uc.hook_add(UC_HOOK_CODE, self.hook_uart_print, begin=0x100ac, end=0x100ac)

        udbserver(self.uc, 1234, self.START_ADDR)

        try:
            # Start emulation from _start function
            self.uc.emu_start(self.START_ADDR, 0)  # 10 second timeout

        except UcError as e:
            pc = self.uc.reg_read(UC_ARM_REG_PC)
            print(f"[-] Emulation error at PC: 0x{pc:08x}")
            print(f"[-] Error: {e} (errno: {e.errno})")


def main():
    if len(sys.argv) != 4:
        print("Usage: python3 emulate_arm.py <text_file> <data_file> <firmware_file>")
        sys.exit(1)

    text_path = sys.argv[1]
    data_path = sys.argv[2]
    firmware_path = sys.argv[3]

    print("ARM Firmware Emulator with UART Capture")
    print("=" * 40)

    emulator = ARMEmulator(text_path, data_path, firmware_path)
    emulator.setup_memory()
    emulator.load_binary()
    emulator.setup_registers()
    emulator.emulate()

if __name__ == "__main__":
    main()
